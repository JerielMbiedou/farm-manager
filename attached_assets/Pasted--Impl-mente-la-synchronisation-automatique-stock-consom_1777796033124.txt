  // Synchronisation automatique avec le stock
  const syncEnabled = await getParam(
    "sync_stock_consommation"
  );

  if (syncEnabled === "true") {
    try {
      // Chercher la désignation la plus proche
      // dans le stock aliments
      const designation = body.typeAliment
        ? typeAlimentToDesignation(body.typeAliment)
        : body.designation ?? "Aliment bande";

      await db.insert(stockAlimentsTable).values({
        designation,
        type: "sortie",
        quantiteKg: String(body.quantiteKg),
        date: body.date
           ?? new Date().toISOString().split("T")[0],
        commentaire: `Auto — ${bande.nom} `
                   + `J${body.ageJours}`,
        fournisseur: null,
        prixUnitaire: null,
      });

      req.log.info({
        bandeId: bande.id,
        quantiteKg: body.quantiteKg,
        designation,
      }, "Sync stock ← consommation bande");

    } catch (err) {
      // Ne pas bloquer la saisie si la sync échoue
      req.log.warn({ err },
        "Sync stock échouée — non bloquant"
      );
    }
  }

Fonction helper typeAlimentToDesignation :
  function typeAlimentToDesignation(
    type: string
  ): string {
    const map: Record<string, string> = {
      demarrage:  "Aliment démarrage",
      croissance: "Aliment croissance",
      finition:   "Aliment finition",
    };
    return map[type] ?? "Aliment bande";
  }

1.3 — Marquer les sorties auto dans le stock
Les sorties créées automatiquement doivent être
distinguables des saisies manuelles.

Ajouter si absent dans stockAlimentsTable le champ :
  source: varchar("source", { length: 50 })
    .default("manuel")
  // Valeurs : "manuel" | "auto_bande"

Si migration nécessaire :
  ALTER TABLE stock_aliments
  ADD COLUMN source varchar(50) DEFAULT 'manuel';

Pour les sorties auto, passer source: "auto_bande".

Si la migration est trop risquée, utiliser le champ
commentaire existant comme marqueur :
  commentaire: `[AUTO] ${bande.nom} J${ageJours}`

Et détecter côté frontend avec :
  item.commentaire?.startsWith("[AUTO]")

Choisir l'approche la plus simple selon l'état
actuel du schéma.
