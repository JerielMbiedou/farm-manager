  // Middleware auth global — protège tout /api
  app.use("/api", (req, res, next) => {
    // Routes publiques exemptées
    const publicPaths = [
      "/api/auth/login",
      "/api/auth/logout",
      "/api/healthz",
    ];
    if (publicPaths.some(p => req.path === p ||
        req.originalUrl.startsWith(p))) {
      return next();
    }
    // Vérifier la session

        error: "Non authentifié — connexion requise"

    next();

noms de dépendances, SQL interne.

  // Handler d'erreurs global — DOIT être le dernier middleware
  app.use((err: Error, req: Request, res: Response,
           next: NextFunction) => {
    // Logger l'erreur complète côté serveur

      method: req.method }, "Erreur serveur non gérée");

FIX-004 — Validation Zod sur toutes les routes d'écriture

PROBLÈME : Mauvais champs ou valeurs négatives acceptés
→ HTTP 500 au lieu de HTTP 400.

Créer un middleware de validation dans
artifacts/api-server/src/lib/validate.ts :

  import { ZodSchema, ZodError } from "zod";
  import { Request, Response, NextFunction } from "express";

  export function validateBody(schema: ZodSchema) {
    return (req: Request, res: Response, next: NextFunction) => {
      try {
        req.body = schema.parse(req.body);
        next();
      } catch (err) {
        if (err instanceof ZodError) {
          return res.status(400).json({
            error: "Données invalides",
            details: err.errors.map(e => ({
              champ: e.path.join("."),
              message: e.message,
            })),
          });
        }
        next(err);
      }
    };
  }

FIX-006 — Corriger l'algorithme de numéro de bande

Dans bandes.ts route POST /, remplacer :
  const numero = (await db.select()
    .from(bandesTable)).length + 1;

Par :
  const maxNumero = await db
    .select({ max: sql`MAX(numero)` })
    .from(bandesTable);
  const numero = (maxNumero[0]?.max ?? 0) + 1;

Ainsi même après suppression d'une bande,
le prochain numéro est toujours unique.
