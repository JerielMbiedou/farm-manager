3.1 — CORS configurable par variable d'environnement
Dans artifacts/api-server/src/app.ts :
  const allowedOrigins = process.env.FRONTEND_URL
    ? process.env.FRONTEND_URL.split(",").map(s => s.trim())
    : ["http://localhost:5173"];

  app.use(cors({
    origin: (origin, callback) => {
      // Autoriser les requêtes sans origin (mobile, curl)
      if (!origin) return callback(null, true);
      if (allowedOrigins.includes(origin) ||
          allowedOrigins.includes("*")) {
        return callback(null, true);
      }
      callback(new Error(`Origin ${origin} non autorisée`));
    },
    credentials: true,
  }));

3.2 — Servir le frontend depuis Express en production
Dans app.ts, ajouter APRÈS les routes API :
  import path from "path";
  import { fileURLToPath } from "url";
  const __dirname = path.dirname(fileURLToPath(import.meta.url));

  if (process.env.NODE_ENV === "production") {
    const frontendDist = path.resolve(
      __dirname,
      "../../../artifacts/ferme-familiale/dist"
    );
    app.use(express.static(frontendDist));
    // Fallback SPA (React Router / wouter)
    app.get("*", (req, res) => {
      if (!req.path.startsWith("/api")) {
        res.sendFile(
          path.join(frontendDist, "index.html"),
          (err) => {
            if (err) res.status(404).send("Not found");
          }
        );
      }
    });
  }
