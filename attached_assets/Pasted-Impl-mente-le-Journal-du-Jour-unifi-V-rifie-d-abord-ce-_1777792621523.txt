  function formatDateFR(dateStr: string): string {
    return new Date(dateStr).toLocaleDateString("fr-FR", {
      weekday: "long",
      day: "numeric",
      month: "long",
      year: "numeric",
    });
  }

── LOGIQUE SUBMIT ──────────────────────────────────

  const onSubmit = async (values) => {
    setIsSubmitting(true);
    const baseUrl = `/api/bandes/${bande.id}`;

    // Vérifier si offlineFetch existe,
    // sinon utiliser fetch classique avec fallback
    const fetchFn = typeof offlineFetch !== "undefined"
      ? offlineFetch
      : async (url, body, desc) => {
          const res = await fetch(url, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(body),
            credentials: "include",
          });
          return { queued: false, data: await res.json() };
        };

    const promises = [];
    let anyQueued = false;

    // Mortalité — toujours envoyée (même 0)
    promises.push(
      fetchFn(
        `${baseUrl}/mortalite`,
        {
          date: today,
          ageJours,
          decesJour: values.deces,
        },
        `Mortalité ${bande.nom} J${ageJours}`
      ).then(r => { if (r.queued) anyQueued = true; })
    );

    // Aliment — seulement si > 0
    if (values.alimentKg && values.alimentKg > 0) {
      promises.push(
        fetchFn(
          `${baseUrl}/consommation`,
          {
            date: today,
            ageJours,
            quantiteKg: values.alimentKg,
            typeAliment: values.typeAliment,
          },
          `Aliment ${bande.nom} J${ageJours}`
        ).then(r => { if (r.queued) anyQueued = true; })
      );
    }

    // Eau — seulement si > 0
    if (values.eauLitres && values.eauLitres > 0) {
      promises.push(
        fetchFn(
          `${baseUrl}/consommation-eau`,
          {
            date: today,
            ageJours,
            quantiteLitres: values.eauLitres,
          },
          `Eau ${bande.nom} J${ageJours}`
        ).then(r => { if (r.queued) anyQueued = true; })
      );
    }

    // Observations — seulement si non vide
    if (values.observations?.trim()) {
      promises.push(
        fetchFn(
          `${baseUrl}/observations`,
          {
            date: today,
            ageJours,
            contenu: values.observations.trim(),
          },
          `Observation ${bande.nom} J${ageJours}`
        ).then(r => { if (r.queued) anyQueued = true; })
      );
    }

    try {
      await Promise.allSettled(promises);

      if (anyQueued) {
        toast.warning(
          "📴 Saisie enregistrée localement — " +
          "synchronisation automatique à la reconnexion"
        );
      } else {
        toast.success(`✓ Journée J${ageJours} enregistrée`);
      }

      // Invalider les queries react-query
      queryClient.invalidateQueries({
        queryKey: ["bande", bande.id]
      });

      onSuccess?.();
      onOpenChange(false);
      form.reset();

    } catch (err) {
      toast.error("Erreur lors de l'enregistrement");
    } finally {
      setIsSubmitting(false);
    }
  };
