  // Compter les bandes clôturées qui ont utilisé cet actif
  const allocationsRows = await db
    .select({
      fractionUtilisee: bandeActifsTable.fractionUtilisee,
      bandeStatut: bandesTable.statut,
    })
    .from(bandeActifsTable)
    .innerJoin(bandesTable,
      eq(bandeActifsTable.bandeId, bandesTable.id))
    .where(eq(bandeActifsTable.actifId, actif.id));

  const nbBandesUtilisees = allocationsRows
    .filter(r => r.bandeStatut === "terminee")
    .reduce((s, r) =>

  const valeur = parseFloat(actif.valeur);

  );
  const valeurResiduelle = Math.max(0, valeur - amortissementCumule);

  const cashflowMensuel: Record<string, {
    mois: string,
    entrees: number,
    sorties: number,
    solde: number
  }> = {};

  for (const entry of journalEntries) {
    const mois = entry.date.substring(0, 7); // YYYY-MM
    if (!cashflowMensuel[mois]) {
      cashflowMensuel[mois] = {
        mois, entrees: 0, sorties: 0, solde: 0
      };
    }
    if (entry.type === "entree") {
      cashflowMensuel[mois].entrees += entry.montant;
    } else {
      cashflowMensuel[mois].sorties += entry.montant;
    }
    cashflowMensuel[mois].solde =
      cashflowMensuel[mois].entrees - cashflowMensuel[mois].sorties;
  }

  const cashflowArray = Object.values(cashflowMensuel)
    .sort((a, b) => a.mois.localeCompare(b.mois));

  // Ajouter cashflowMensuel à la réponse :
