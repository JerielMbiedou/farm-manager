Dans src/lib/use-sortable-table.ts, créer :
  export function useSortableTable<T>(
    data: T[],
    pageSize = 50
  ) {
    const [sortKey, setSortKey] = useState<keyof T | null>(null);
    const [sortDir, setSortDir] = useState<"asc" | "desc">("asc");
    const [page, setPage] = useState(1);

    const sorted = useMemo(() => {
      if (!sortKey) return data;
      return [...data].sort((a, b) => {
        const av = a[sortKey], bv = b[sortKey];
        const cmp = av < bv ? -1 : av > bv ? 1 : 0;
        return sortDir === "asc" ? cmp : -cmp;
      });
    }, [data, sortKey, sortDir]);

    const paginated = sorted.slice(
      (page - 1) * pageSize, page * pageSize
    );
    const totalPages = Math.ceil(sorted.length / pageSize);

    const toggleSort = (key: keyof T) => {
      if (sortKey === key) {
        setSortDir(d => d === "asc" ? "desc" : "asc");
      } else {
        setSortKey(key);
        setSortDir("asc");
      }
      setPage(1);
    };

    return { paginated, sorted, page, setPage,
             totalPages, sortKey, sortDir, toggleSort };
  }

  import { openDB } from "idb";
  const DB_NAME = "ferme-offline-queue";
  const STORE = "requests";

  interface QueuedRequest {
    id: string; url: string; method: string;
    body: any; timestamp: number; description: string;
  }

  async function getDB() {
    return openDB(DB_NAME, 1, {
      upgrade(db) { db.createObjectStore(STORE,{keyPath:"id"}); }
    });
  }

  export async function enqueueRequest(
    url: string, method: string, body: any, description: string
  ) {
    const db = await getDB();
    await db.put(STORE, {
      id: crypto.randomUUID(), url, method, body,
      timestamp: Date.now(), description
    });
  }

  export async function getPendingRequests(): Promise<QueuedRequest[]> {
    return (await getDB()).getAll(STORE);
  }

  export async function removeRequest(id: string) {
    return (await getDB()).delete(STORE, id);
  }

  export async function syncPendingRequests() {
    const pending = await getPendingRequests();
    let success = 0, failed = 0;
    const errors: string[] = [];
    for (const req of pending) {
      try {
        const res = await fetch(req.url, {
          method: req.method,
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(req.body),
          credentials: "include",
        });
        if (res.ok) { await removeRequest(req.id); success++; }
        else { failed++; errors.push(`${req.description}: ${res.status}`); }
      } catch {
        failed++;
        errors.push(`${req.description}: réseau indisponible`);
      }
    }
    return { success, failed, errors };
  }

10.5 — Wrapper postWithOffline
Créer src/lib/api-with-offline.ts :

  import { enqueueRequest } from "./offline-queue";

  export async function postWithOffline(
    url: string, body: any, description: string
  ): Promise<{ queued: boolean; data?: any }> {
    if (!navigator.onLine) {
      await enqueueRequest(url, "POST", body, description);
      return { queued: true };
    }
    try {
      const res = await fetch(url, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(body),
        credentials: "include",
      });
      if (!res.ok) throw new Error(await res.text());
      return { queued: false, data: await res.json() };
    } catch {
      await enqueueRequest(url, "POST", body, description);
      return { queued: true };
    }
  }
