5.1 — Créer l'utilitaire si absent

  export function getAgeJours(dateDeDepart: string): number {
    const start = new Date(dateDeDepart + "T00:00:00");
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    return Math.max(
      1,
      Math.floor(
        (today.getTime() - start.getTime()) / 86400000
      ) + 1
    );
  }
