BLOC 9 — INTÉGRITÉ DES DONNÉES EN DB
═══════════════════════════════════════════════════

T9.1 — Vérifier orphelins potentiels
  Exécuter ces requêtes SQL directement :

  -- Mortalités sans bande
  SELECT COUNT(*) FROM mortalite_journaliere m
  LEFT JOIN bandes b ON m.bande_id = b.id
  WHERE b.id IS NULL;

  -- Vaccinations sans bande
  SELECT COUNT(*) FROM vaccinations v
  LEFT JOIN bandes b ON v.bande_id = b.id
  WHERE b.id IS NULL;

  -- Actifs sans chantier (si chantier_id renseigné)
  SELECT COUNT(*) FROM actifs a
  WHERE a.chantier_id IS NOT NULL
  AND NOT EXISTS (
    SELECT 1 FROM chantiers c WHERE c.id = a.chantier_id
  );

  -- Dépenses sans bande
  SELECT COUNT(*) FROM bande_depenses bd
  LEFT JOIN bandes b ON bd.bande_id = b.id
  WHERE b.id IS NULL;

  → Tous doivent retourner 0

T9.2 — Vérifier cohérence sujets vivants
  Pour chaque bande active :
  SELECT b.sujets_depart, b.nombre_deces,
    (SELECT COALESCE(SUM(quantite_vendue), 0)
     FROM bande_ventes WHERE bande_id = b.id) as vendus,
    b.sujets_depart - b.nombre_deces -
    (SELECT COALESCE(SUM(quantite_vendue), 0)
     FROM bande_ventes WHERE bande_id = b.id) as calcule
  FROM bandes b WHERE b.statut = 'active';
  → calcule doit correspondre à ce qu'affiche l'UI

T9.3 — Vérifier cohérence financière
  -- Total financement
  SELECT SUM(montant) FROM financement;
  -- Total remboursements
  SELECT SUM(montant) FROM remboursements;
  -- Dépenses chantiers (nouveau système)
  SELECT SUM(quantite * prix_unitaire)
  FROM chantier_depenses;
  -- Vérifier que ces chiffres correspondent au dashboard

T9.4 — Vérifier dépenses sans date
  SELECT COUNT(*) FROM bande_depenses WHERE date IS NULL;
  → Idéalement 0 (ou noter le nombre pour backfill)

T9.5 — Vérifier index dupliqué session
  SELECT indexname FROM pg_indexes
  WHERE tablename = 'session'
  ORDER BY indexname;
  → Ne doit pas y avoir 2 index sur expire

T9.6 — Vérifier FK vaccinations
  SELECT COUNT(*) FROM vaccinations v
  WHERE NOT EXISTS (
    SELECT 1 FROM bandes b WHERE b.id = v.bande_id
  );
  → Doit retourner 0
